## Read-only investigation
```sql
SELECT now() AT TIME ZONE 'UTC' AS checked_utc,
       pg_database_size(current_database()) AS db_bytes,
       current_setting('max_connections')::int AS max_connections,
       (SELECT count(*) FROM pg_stat_activity) AS used_connections;
SELECT n.nspname, c.relname, pg_total_relation_size(c.oid) AS total_bytes,
       pg_relation_size(c.oid) AS heap_bytes,
       pg_indexes_size(c.oid) AS index_bytes
FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace
WHERE c.relkind IN ('r','m') AND n.nspname NOT IN ('pg_catalog','information_schema')
ORDER BY total_bytes DESC LIMIT 25;
SELECT jobid,jobname,schedule,active,command FROM cron.job ORDER BY jobid;
SELECT status,count(*),min(queued_at) AS oldest_queued_at
FROM public.origination_reprocessing_queue GROUP BY status;
SELECT jobid,status,count(*) AS runs,max(end_time) AS latest_end
FROM cron.job_run_details WHERE start_time > now()-interval '2 hours'
GROUP BY jobid,status ORDER BY jobid,status;
SELECT pid,now()-xact_start AS age,wait_event_type,wait_event,state
FROM pg_stat_activity WHERE xact_start IS NOT NULL
ORDER BY xact_start LIMIT 20;
```
